cmake_minimum_required(VERSION 3.20)
project(C_Tetris VERSION 1.0.0 LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
option(TETRIS_BUILD_TESTS "Build the headless gameplay tests" ON)
option(TETRIS_FETCH_RAYLIB "Fetch Raylib when no installed package is found" ON)
add_library(tetris_core src/board.c src/tetromino.c src/game.c)
target_include_directories(tetris_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
if(MSVC)
  target_compile_options(tetris_core PRIVATE /W4)
else()
  target_compile_options(tetris_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(TETRIS_BUILD_TESTS)
  enable_testing()
  add_executable(tetris_tests tests/test_game.c)
  target_link_libraries(tetris_tests PRIVATE tetris_core)
  add_test(NAME gameplay COMMAND tetris_tests)
endif()
find_package(raylib 5.0 QUIET)
if(NOT raylib_FOUND AND TETRIS_FETCH_RAYLIB)
  include(FetchContent)
  set(BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
  set(BUILD_GAMES OFF CACHE BOOL "" FORCE)
  set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
  FetchContent_Declare(raylib GIT_REPOSITORY https://github.com/raysan5/raylib.git GIT_TAG 5.5 GIT_SHALLOW TRUE)
  FetchContent_MakeAvailable(raylib)
endif()
if(raylib_FOUND OR TARGET raylib)
  add_executable(c_tetris src/main.c src/renderer.c src/input.c src/effects.c src/audio.c)
  target_include_directories(c_tetris PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/include)
  target_link_libraries(c_tetris PRIVATE tetris_core raylib)
  if(WIN32)
    target_link_libraries(c_tetris PRIVATE winmm)
  elseif(NOT MSVC)
    target_link_libraries(c_tetris PRIVATE m)
  endif()
  if(MSVC)
    target_compile_options(c_tetris PRIVATE /W4)
  else()
    target_compile_options(c_tetris PRIVATE -Wall -Wextra -Wpedantic)
  endif()
else()
  message(WARNING "Raylib not found. Headless tests can still build; install Raylib or enable TETRIS_FETCH_RAYLIB for the game executable.")
endif()
